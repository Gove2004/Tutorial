第八章：FFI 与 C 互操作

Rust 的一大优势是可以与现有的 C 生态系统无缝集成。通过 FFI（Foreign Function Interface），Rust 可以直接调用 C 函数、链接 C 库、甚至被 C 程序调用。本章将全面介绍 Rust 的 FFI 机制、自动绑定生成工具和安全的互操作模式。

为什么 FFI 对系统编程至关重要？

世界上绝大部分的系统级代码是用 C 编写的：Linux 内核、数据库引擎、图形库、网络协议栈……Rust 的 FFI 让你可以渐进式迁移——不必重写整个系统，而是在关键路径上用 Rust 替换 C 代码。

8.1 extern "C"：调用 C 函数的精确语义

extern "C" 块声明外部函数的签名，告诉编译器这些函数在别处实现，并使用 C 语言的 ABI（调用约定）：

// 直接声明 C 函数
extern "C" {
    fn printf(format: *const i8, ...) -> i32;
    fn malloc(size: usize) -> *mut u8;
    fn free(ptr: *mut u8);
    fn strlen(s: *const i8) -> usize;
    fn perror(s: *const i8);
}

// 调用示例
unsafe {
    let msg = b"Hello from Rust!\0" as *const u8 as *const i8;
    printf(msg);
}
危险警告：调用 extern 函数必须在 unsafe 块中进行，因为这些调用绕过了 Rust 的安全性检查（如借用检查、空指针检查、数据竞争检查）。确保你理解每个 C 函数的内存管理和生命周期要求。

8.1.1 ABI 差异

Rust 支持多种 ABI，不同 ABI 的调用约定不同：

ABI 用途 参数传递 返回值
"C" 与 C 互操作 寄存器/栈（平台相关） 寄存器
"cdecl" C 默认调用约定 栈（调用者清理） eax
"stdcall" Windows API 栈（被调用者清理） eax
"system" 平台默认 自动选择 自动选择
"efiapi" UEFI 固件 同 stdcall 同 stdcall

8.1.2 C 字符串的处理

use std::ffi::{CStr, CString};
use std::os::raw::c_char;

// 将 Rust &str 传给 C 函数
fn greet(name: &str) {
    // CString 将 &str 转换为 '\0' 结尾的 C 字符串
    let c_name = CString::new(name).expect("内含 NUL 字节");

    unsafe {
        // 调用 C 的 puts 函数
        libc::puts(c_name.as_ptr());
    }
}

// 从 C 函数接收 &str
fn get_greeting() -> Result<String, &'static str> {
    unsafe {
        // 假设 C 函数返回 const char*
        let ptr = libc::gets(c_buffer.as_mut_ptr());
        if ptr.is_null() {
            return Err("NULL pointer from C");
        }
        // CStr::from_ptr 创建零拷贝引用
        CStr::from_ptr(ptr)
            .to_str()
            .map(|s| s.to_string())
            .map_err(|_| "invalid UTF-8 from C")
    }
}

8.2 bindgen：自动生成 FFI 绑定

bindgen 是 Rust 官方推荐的工具，可以从 C 头文件自动生成安全的 Rust 绑定。它解析 C 头文件并生成对应的 Rust 声明：

// Cargo.toml
// [build-dependencies]
// bindgen = "0.70"

// build.rs
fn main() {
    let bindings = bindgen::builder()
        .header("lib/mylib.h")
        .clang_arg("-I/usr/include")
        .layout_tests(true)
        .generate()
        .expect("Unable to generate bindings");

    bindings
        .write_to_file("src/bindings.rs")
        .expect("Couldn't write bindings!");
}

8.2.1 生成的绑定示例

// 原始 C 头文件 (mylib.h)：
// typedef struct { int x; double y; } Point;
// Point create_point(int x, double y);
// double distance(Point a, Point b);

// bindgen 生成的 Rust 代码：
#[repr(C)]
#[derive(Debug, Copy, Clone)]
pub struct Point {
    pub x: i32,
    pub y: f64,
}

#[test]
fn bindgen_test_layout_Point() {
    assert_eq!(
        std::mem::size_of::<Point>(), 16,
        "Size of Point"
    );
    assert_eq!(
        std::mem::align_of::<Point>(), 8,
        "Alignment of Point"
    );
}

extern "C" {
    pub fn create_point(x: i32, y: f64) -> Point;
}
extern "C" {
    pub fn distance(a: Point, b: Point) -> f64;
}

8.2.2 bindgen 的高级选项

bindgen::builder()
    .header("lib.h")
    .allowlist_type("MyStruct")       // 只生成指定的类型
    .allowlist_function("my_func")    // 只生成指定的函数
    .allowlist_var("MY_CONST")        // 只生成指定的常量
    .blocklist_type("FILE")           // 跳过 FILE（不透明类型）
    .opaque_type("FILE")              // FILE 作为不透明类型
    .ctypes_prefix("std::ffi")        // 使用 std::ffi 的 c_uint 等
    .rustfmt_bindings(true)           // 格式化输出
    .generate_cstr(true)              // const char* 转为 &CStr
    .bitfield_style(bindgen::BitfieldStyle::Rust)  // 位域风格
    .default_enum_style(bindgen::EnumVariation::Rust)  // 枚举风格
    .derive_debug(true)               // 自动生成 Debug
    .derive_default(true)             // 自动生成 Default
    .derive_hash(true)                // 自动生成 Hash
    .generate()
性能提示：bindgen 在编译时运行，只增加构建时间。生成的绑定代码是纯声明，没有运行时开销。对于大型 C 库（如 OpenSSL），bindgen 生成的绑定文件可能有数万行，但编译时间仅增加 5-15 秒。

8.3 cbindgen：从 Rust 生成 C 头文件

如果你用 Rust 编写库并希望被 C 程序调用，cbindgen 可以自动生成 C 头文件：

// Cargo.toml
// [build-dependencies]
// cbindgen = "0.27"

// build.rs
fn main() {
    cbindgen::Builder::new()
        .with_crate(".")
        .with_config(cbindgen::Config::from_file("cbindgen.toml").unwrap())
        .generate()
        .and_then(|bindings| bindings.write_to_file("include/mylib.h"));
}
// cbindgen.toml
[export]
include = ["MyStruct", "my_function"]
prefix = "MYLIB_"

[ptr]
safe_any = true

[enum]
prefix = "MYLIB_ENUM_"

[compile]
clean = true  // 每次重新生成

[c.include]
"stdint.h"
"stdlib.h"

[c.macro_alternate]
"MYLIB_NULL" = "NULL"

8.3.1 从 Rust 导出给 C 调用的示例

// lib.rs — 用 Rust 编写的库
#[repr(C)]
pub struct MyStruct {
    pub value: i32,
    pub name: *const i8,  // C 字符串指针
}

/// 创建一个新实例
#[no_mangle]
pub extern "C" fn mylib_create(value: i32, name: *const i8) -> *mut MyStruct {
    let name_str = unsafe {
        std::ffi::CStr::from_ptr(name).to_str().unwrap()
    };
    let s = Box::new(MyStruct {
        value,
        name: name.as_ref().map(|s| s.as_ptr()).unwrap_or(std::ptr::null()),
    });
    Box::into_raw(s)
}

/// 释放内存
#[no_mangle]
pub extern "C" fn mylib_destroy(s: *mut MyStruct) {
    if !s.is_null() {
        unsafe { drop(Box::from_raw(s)) };
    }
}

/// 获取值
#[no_mangle]
pub extern "C" fn mylib_get_value(s: *const MyStruct) -> i32 {
    unsafe { (*s).value }
}

8.4 安全边界：Safe Wrappers 模式

FFI 的核心原则：unsafe 代码尽量靠近边界，safe 代码尽量靠近核心。为每个 FFI 函数编写 safe wrapper：

use std::ffi::{CStr, CString};
use std::os::raw::c_char;

// 不安全的原始 FFI 声明
extern "C" {
    fn c_strlen(s: *const c_char) -> usize;
    fn c_strdup(s: *const c_char) -> *mut c_char;
    fn c_free(ptr: *mut c_char);
}

// Safe wrapper 1：strlen
pub fn safe_strlen(s: &str) -> usize {
    let c_str = CString::new(s).expect("string contains nul byte");
    unsafe { c_strlen(c_str.as_ptr()) }
}

// Safe wrapper 2：strdup — 使用 RAII 管理 C 分配的内存
pub struct CStringWrapper {
    ptr: *mut c_char,
}

impl CStringWrapper {
    pub fn new(s: &str) -> Result<Self, std::ffi::NulError> {
        let c_str = CString::new(s)?;
        let copied = unsafe { c_strdup(c_str.as_ptr()) };
        if copied.is_null() {
            return Err(std::ffi::NulError::new(0, vec![]));
        }
        Ok(Self { ptr: copied })
    }

    pub fn as_str(&self) -> &str {
        unsafe {
            CStr::from_ptr(self.ptr)
                .to_str()
                .expect("invalid UTF-8")
        }
    }
}

impl Drop for CStringWrapper {
    fn drop(&mut self) {
        if !self.ptr.is_null() {
            unsafe { c_free(self.ptr) };
            self.ptr = std::ptr::null_mut();
        }
    }
}

// 使用：自动管理 C 内存
fn main() {
    let wrapped = CStringWrapper::new("hello world").unwrap();
    println!("Length: {}", safe_strlen(wrapped.as_str()));
    println!("Content: {}", wrapped.as_str());
    // wrapped 离开作用域时自动 free
}

FFI 安全边界架构

┌─────────────────────────────────────────────┐
│          Safe Application Code              │  ← 零 unsafe
│  fn process_config(path: &str) -> Result<…> │
│    let raw = ffi_load_file(path)?;           │
│    let parsed = parse_config(raw.as_bytes()); │

└──────────────────┬──────────────────────────┘

                   ▼

│         Safe FFI Wrapper Layer              │  ← 少量 unsafe
│  fn ffi_load_file(path: &str) -> RawBuffer  │
│    let c_path = CString::new(path)?;         │

                   ▼

8.5 ABI 兼容性详解

8.5.1 repr(C) 结构体内存布局

#[repr(C)] 确保 Rust 结构体的内存布局与 C 兼容：

// 默认布局：Rust 可以自由重排字段以优化内存
struct DefaultLayout {
    a: u8,    // 1 byte
    b: u32,   // 4 bytes
    c: u8,    // 1 byte
}
// 大小：12 字节（Rust 可能重排字段）

// C 兼容布局：字段按声明顺序排列
#[repr(C)]
struct CCompatible {
    a: u8,
    _padding: [u8; 3],  // 手动填充到 4 字节对齐
    b: u32,
    c: u8,
    _padding2: [u8; 3],
}
// 大小：12 字节（与 C 结构体一致）

// 紧凑布局（无填充，但可能产生未对齐访问）
#[repr(C, packed)]
struct Packed {
    a: u8,
    b: u32,
    c: u8,
}
// 大小：6 字节（无填充）
// 注意：packed 结构体的未对齐访问可能导致性能下降
性能警告：packed 结构体会导致未对齐内存访问，在现代 CPU 上可能触发微码级别的惩罚（约 2-3x slowdown）。除非与 C 库的 binary 格式严格兼容，否则尽量避免使用 packed。

8.5.2 枚举的 C 兼容

// 指定底层表示类型
#[repr(u8)]
enum Color {
    Red = 0,
    Green = 1,
    Blue = 2,
}

// 等价于 C 代码：
// typedef enum { RED = 0, GREEN = 1, BLUE = 2 } Color;

fn c_print_color(color: Color) {
    println!("Color value: {}", color as u8);
}

// 处理 C 传来的未知枚举值
#[repr(u8)]
enum Status {
    Ok = 0,
    Error = 1,
    Timeout = 2,
}

impl Status {
    // from_u8 处理 C 传来的未知值
    pub fn from_u8(v: u8) -> Option<Self> {
        match v {
            0 => Some(Status::Ok),
            1 => Some(Status::Error),
            2 => Some(Status::Timeout),
            _ => None,  // C 传来的未知值
        }
    }
}

8.5.3 Bitfields 的处理

// C 中的位字段
// struct Flags {
//     unsigned enabled : 1;
//     unsigned readonly : 1;
//     unsigned hidden : 1;
//     unsigned reserved : 29;
// };

#[repr(C)]
#[derive(Default)]
struct Flags {
    bits: u32,
}

impl Flags {
    pub fn enabled(&self) -> bool {
        self.bits & 0x1 != 0
    }

    pub fn set_enabled(&mut self, value: bool) {
        self.bits = (self.bits & !0x1) | ((value as u32) & 0x1);
    }

    pub fn readonly(&self) -> bool {
        (self.bits & 0x2) != 0
    }

    pub fn hidden(&self) -> bool {
        (self.bits & 0x4) != 0
    }
}

8.6 从 C 调用 Rust

Rust 也可以被 C 程序调用。关键在于使用 #[no_mangle] 和 extern "C"：

// Rust 端 (lib.rs)
use std::os::raw::c_int;

/// 计算两个整数的和
#[no_mangle]
pub extern "C" fn rust_add(a: c_int, b: c_int) -> c_int {
    a + b
}

/// 处理 C 字符串
#[no_mangle]
pub extern "C" fn rust_reverse_string(input: *const i8) -> *mut i8 {
    unsafe {
        let c_str = CStr::from_ptr(input);
        let s = c_str.to_str().unwrap();
        let reversed: String = s.chars().rev().collect();
        CString::new(reversed).unwrap_into_raw()
    }
}

// C 端调用示例：
// #include "rust_lib.h"
// int main() {
//     int result = rust_add(3, 4);    // 7
//     char* rev = rust_reverse_string("hello");
//     printf("%s\n", rev);            // "olleh"
//     free(rev);                      // 需要 C 端 free
//     return 0;

8.7 FFI 性能优化

// FFI 调用的性能考量

// 1. 减少 FFI 调用次数（批量处理）
// 差：逐个调用（1000 次 FFI 调用 = ~100μs）
for item in items {
    unsafe { ffi_process_single(item) };
}

// 好：批量调用（1 次 FFI 调用 = ~1μs）
unsafe {
    ffi_process_batch(items.as_ptr(), items.len() as i32);
}

// 2. 避免不必要的内存拷贝
// 差：每次传递字符串都复制
let c_str = CString::new(rust_string).unwrap();
ffi_accepts_cstr(c_str.as_ptr());  // 创建 CString → 复制

// 好：直接使用 &CStr
let c_str: &CStr = CStr::from_bytes_with_nul(b"hello\0").unwrap();
ffi_accepts_cstr(c_str.as_ptr());  // 零拷贝

// 3. 使用 #[inline] 减少桥接开销
#[no_mangle]
#[inline]
pub extern "C" fn fast_add(a: i32, b: i32) -> i32 {
    a + b
}
场景 性能影响 建议
单次 FFI 调用 ~100-200ns（跨 ABI 边界） 批量处理减少调用次数
频繁 FFI 调用 累积开销显著 使用 bulk API
字符串传递 需要编码转换 使用 &CStr 避免复制
回调函数 Rust → C 回调安全 使用 extern "C" fn

8.8 本章小结

关键要点
  • extern "C"：声明外部函数的基本语法，需要 unsafe 块
  • bindgen：从 C 头文件自动生成 Rust 绑定，支持 allowlist/blocklist
  • cbindgen：从 Rust 生成 C 头文件，支持配置化导出
  • Safety boundary：unsafe 靠近边界，safe 靠近核心，用 RAII 管理资源
  • repr(C)：确保内存布局与 C 兼容，注意 packed 的性能惩罚
  • 批量处理：减少 FFI 调用次数是关键优化手段
  • #[no_mangle]：让 Rust 函数可以被 C 调用
