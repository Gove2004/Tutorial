10.1 定义自定义 Gymnasium 环境

Gymnasium 环境遵循统一接口，只需继承 gymnasium.Env 并实现四个核心方法：reset、step、render 和 close。

import gymnasium as gym
from gymnasium import spaces
import numpy as np
import pygame

class CustomMazeEnv(gym.Env):
    """
    自定义迷宫环境：Agent 需要从起点到达终点
    - 4x4 网格地图
    - 离散动作空间（上下左右）
    - 连续观测空间（Agent 位置 + 终点位置）
    """
    metadata = {"render_modes": ["human", "rgb_array"]}

    def __init__(self, render_mode=None, wall_prob=0.2):
        super().__init__()
        self.render_mode = render_mode
        self.grid_size = 4
        self.wall_prob = wall_prob

        # 定义空间
        self.observation_space = spaces.Box(
            low=np.array([0, 0, 0, 0]),
            high=np.array([self.grid_size-1, self.grid_size-1, self.grid_size-1, self.grid_size-1]),
            dtype=np.float32
        )
        self.action_space = spaces.Discrete(4)  # 0:上 1:下 2:左 3:右

        # 动作映射
        self.action_map = {0: (-1, 0), 1: (1, 0), 2: (0, -1), 3: (0, 1)}

        self._generate_maze()

    def _generate_maze(self):
        """随机生成迷宫（障碍物）"""
        self.walls = set()
        for _ in range(int(self.grid_size * self.grid_size * self.wall_prob)):
            pos = (np.random.randint(self.grid_size), np.random.randint(self.grid_size))
            if pos != (0, 0) and pos != (self.grid_size-1, self.grid_size-1):
                self.walls.add(pos)

    def reset(self, seed=None, options=None):
        super().reset(seed=seed)
        self.agent_pos = (0, 0)
        self.goal_pos = (self.grid_size-1, self.grid_size-1)
        self.steps = 0
        self.max_steps = self.grid_size * self.grid_size * 2

        state = np.array([*self.agent_pos, *self.goal_pos], dtype=np.float32)
        return state, {}

    def step(self, action):
        dr, dc = self.action_map[action]
        new_r = max(0, min(self.grid_size-1, self.agent_pos[0] + dr))
        new_c = max(0, min(self.grid_size-1, self.agent_pos[1] + dc))
        new_pos = (new_r, new_c)

        if new_pos in self.walls:
            reward = -10  # 撞墙
            done = True
        elif new_pos == self.goal_pos:
            reward = 100  # 到达目标
            done = True
        else:
            reward = -1  # 每步小惩罚，鼓励尽快到达
            done = False

        self.agent_pos = new_pos
        self.steps += 1

        if self.steps >= self.max_steps:
            done = True
            reward = -50  # 超时惩罚

        state = np.array([*self.agent_pos, *self.goal_pos], dtype=np.float32)
        return state, reward, done, False, {}

    def render(self):
        if self.render_mode == "human":
            # 简化版：打印网格
            grid = [['.' for _ in range(self.grid_size)] for _ in range(self.grid_size)]
            for r, c in self.walls:
                grid[r][c] = '#'
            grid[self.agent_pos[0]][self.agent_pos[1]] = 'A'
            grid[self.goal_pos[0]][self.goal_pos[1]] = 'G'
            for row in grid:
                print(' '.join(row))
            print()

# 测试环境
env = CustomMazeEnv(render_mode="human")
state, _ = env.reset()
for _ in range(20):
    action = env.action_space.sample()
    state, reward, done, truncated, info = env.step(action)
    if done:
        break
env.close()

10.2 视觉输入：Atari 风格处理

当环境输入是图像时，需要对原始像素进行预处理才能输入神经网络：

import cv2
import torch
from torchvision import transforms

class AtariPreprocessor:
    """Atari 标准预处理流程"""
    def __init__(self, frame_height=84, frame_width=84, n_frames=4, gray=True):
        self.frame_height = frame_height
        self.frame_width = frame_width
        self.n_frames = n_frames
        self.gray = gray
        self.frame_buffer = []
        self.skip = 4  # 每 4 帧采样一次

    def __call__(self, frame):
        # 1. 灰度化
        if self.gray:
            frame = cv2.cvtColor(frame, cv2.COLOR_RGB2GRAY)
        # 2. 下采样
        frame = cv2.resize(frame, (self.frame_width, self.frame_height),
                          interpolation=cv2.INTER_AREA)
        # 3. 二值化（可选，Atari 常用）
        frame = (frame > 0).astype(np.float32) * 255
        return frame

    def reset(self, frame):
        self.frame_buffer = []
        processed = self(frame)
        self.frame_buffer = [processed] * self.n_frames
        return self._get_observation()

    def step(self, frame):
        processed = self(frame)
        self.frame_buffer.append(processed)
        if len(self.frame_buffer) > self.n_frames:
            self.frame_buffer.pop(0)
        return self._get_observation()

    def _get_observation(self):
        """堆叠 n_frames 作为单次观测"""
        return np.stack(self.frame_buffer, axis=-1)  # (84, 84, 4)

class VisionCNN(nn.Module):
    """处理图像输入的策略网络"""
    def __init__(self, action_dim, input_channels=4):
        super().__init__()
        self.cnn = nn.Sequential(
            nn.Conv2d(input_channels, 32, kernel_size=8, stride=4),
            nn.ReLU(),
            nn.Conv2d(32, 64, kernel_size=4, stride=2),
            nn.ReLU(),
            nn.Conv2d(64, 64, kernel_size=3, stride=1),
            nn.ReLU()
        )
        # 计算 flattened 大小
        self.fc = nn.Sequential(
            nn.Linear(64 * 7 * 7, 512),
            nn.ReLU(),
            nn.Linear(512, action_dim)
        )

    def forward(self, x):
        # x: (batch, 4, 84, 84)
        x = self.cnn(x)
        x = x.view(x.size(0), -1)
        return self.fc(x)

            })

    def select_actions(self, states):
        """所有 Agent 同时选择动作"""
        actions, log_probs, values = [], [], []
        for i, (state, agent) in enumerate(zip(states, self.agents)):

10.6 本章小结

  • 自定义 Gymnasium 环境只需继承 Env 并实现 reset/step/render/close
  • 视觉输入需要预处理（灰度化、下采样、帧堆叠）
  • 多智能体 RL 的核心挑战是非稳态环境
  • ML-Agents 通过 gRPC 实现 Python-Unity 通信
  • 多进程并行环境采集可显著提升样本效率
