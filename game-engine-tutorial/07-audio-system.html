第七章：音频系统——空间音频、混音与程序化生成

学习目标：理解游戏引擎音频架构的全貌——从 FMOD/Wwise 集成、3D 空间音频、混音效果到程序化声音生成。音频占游戏总内存的 10-30%，但直接影响沉浸感的 50% 以上。本章将深入引擎底层的音频管线设计。

FMOD 集成：加载、播放、停止

FMOD 和 Wwise 是游戏行业两大音频中间件。它们提供跨平台、高性能的音频 API，同时封装了复杂的 DSP 管线。这里以 FMOD 为例。

#include <fmod.hpp>
#include <fmod_common.h>

class AudioSystem {
public:
    void initialize() {
        // 创建 FMOD 系统
        FMOD::System_Create(&m_system);
        m_system->init(32, FMOD_INIT_NORMAL, nullptr);

        // 创建主混音组
        m_system->createChannelGroup("Master", &m_masterGroup);
        m_masterGroup->setVolume(1.0f);
    }

    // 加载音频片段
    void loadSound(const std::string& path, FMOD_MODE mode) {
        FMOD_SOUND* sound = nullptr;
        m_system->createSound(path.c_str(), mode, nullptr, &sound);
        m_sounds[path] = sound;
    }

    // 播放 2D 音频（UI 音效、背景音乐）
    FMOD_CHANNEL* play2D(const std::string& soundName, bool loop = false) {
        FMOD_CHANNEL* channel = nullptr;
        auto it = m_sounds.find(soundName);
        if (it == m_sounds.end()) return nullptr;

        it->second->setLoop(loop, 1);  // 设置循环
        m_system->playSound(it->second, nullptr, false, &channel);
        m_channelMap[channel] = soundName;
        return channel;
    }

    // 停止所有音频
    void stopAll() {
        for (auto& [ch, name] : m_channelMap) {
            ch->stop();
        }
        m_channelMap.clear();
    }

    // 设置全局音量
    void setMasterVolume(float volume) {
        m_masterGroup->setVolume(volume);
    }

    // 每帧更新 FMOD（必须调用）
    void update() {
        m_system->update();
        cleanupStoppedChannels();
    }

private:
    FMOD::System* m_system = nullptr;
    FMOD::ChannelGroup* m_masterGroup = nullptr;
    std::unordered_map<std::string, FMOD::Sound*> m_sounds;
    std::unordered_map<FMOD::Channel*, std::string> m_channelMap;

    void cleanupStoppedChannels() {
        std::vector<FMOD::Channel*> toRemove;
        for (auto& [ch, name] : m_channelMap) {
            bool playing = false;
            ch->isPlaying(&playing);
            if (!playing) toRemove.push_back(ch);
        }
        for (auto ch : toRemove) {
            m_channelMap.erase(ch);
        }
    }
};

3D 空间音频：距离衰减与多普勒效应

3D 音频让声音具有方向性和距离感。玩家听到身后脚步声时的音量应该比前方小，而且随着距离增大而衰减。

距离衰减模型

// 3D 音频：距离衰减 + 多普勒效应
void play3DSound(AudioSystem& audio, FMOD::Sound* sound,
                 const Vec3& listenerPos, const Vec3& emitterPos,
                 const Vec3& emitterVelocity) {
    FMOD_CHANNEL* channel = nullptr;

    // 播放 3D 声音
    audio.system->playSound(sound, nullptr, true, &channel);

    // 设置 3D 位置
    FMOD_VECTOR pos = {emitterPos.x, emitterPos.y, emitterPos.z};
    channel->set3DAttributes(&pos, nullptr);

    // 配置衰减模型
    channel->setMode(FMOD_LOOP_NORMAL);
    channel->setVolume(1.0f);

    // 自定义衰减曲线
    FMOD_3D_ATTRIBUTES attrs;
    attrs.position = pos;
    attrs.velocity = {emitterVelocity.x, emitterVelocity.y, emitterVelocity.z};
    attrs.listenerGain = 1.0f;
    attrs.internalGain = 1.0f;
    attrs.extMin = 1.0f;    // 最小距离（满音量）
    attrs.extMax = 50.0f;   // 最大距离（静音）
    channel->set3DAttributes(&attrs, nullptr);
}

三种衰减模型

模型 公式 特点 适用场景
ReverseV = 1.0 / d距离越近音量越大，无上限近距离对话
LinearV = max(0, 1 - d/maxDist)线性衰减，简单直观通用游戏音效
ExponentialV = 1 / (1 + d² * rollOff)自然衰减，符合物理环境音、音乐

多普勒效应

// 多普勒频移：声源靠近时音调升高，远离时降低
// f' = f * (v_sound + v_listener) / (v_sound - v_emitter)
float dopplerShift(float sourceFreq, float soundSpeed,
                   float listenerVel, float emitterVel) {
    // listenerVel 和 emitterVel 是沿连线方向的速度分量
    float observedFreq = sourceFreq * (soundSpeed + listenerVel)
                                   / (soundSpeed - emitterVel);
    return observedFreq;
}

// 实际应用中，FMOD 内部自动处理多普勒
// 你只需要设置声源和听众的速度属性
void setupDoppler(FMOD::Channel* channel, const Vec3& listenerVel,
                  const Vec3& emitterVel) {
    FMOD_3D_ATTRIBUTES attrs;
    attrs.velocity = {emitterVel.x, emitterVel.y, emitterVel.z};
    // listener velocity 通过 set3DAttributes 的第二个参数传入
    channel->set3DAttributes(&attrs, nullptr);
}

混音系统：音量、声像、均衡与混响

每个音频通道（Voice）都可以独立设置音量、左右声道平衡（Pan）、均衡（EQ）和效果（Reverb/Delay）。混音器将这些通道组合成总线。

class MixerBus {
public:
    // 创建混音总线
    MixerBus(const std::string& name, int numChannels = 2)
        : m_name(name), m_numChannels(numChannels) {
        m_buffer.resize(BUFFER_SIZE * numChannels);
        m_volume = 1.0f;
    }

    // 添加音频源到总线
    void addSource(AudioSource* source) {
        m_sources.push_back(source);
    }

    // 混音：将所有源的 PCM 数据相加
    void mix(float* output, float deltaTime) {
        // 清零输出缓冲区
        std::fill(output, output + m_numChannels * BUFFER_FRAMES, 0.0f);

        // 累加所有源
        for (auto* source : m_sources) {
            if (!source->isPlaying()) continue;

            // 获取源的 PCM 数据
            std::vector<float> sourcePCM = source->getPCM(deltaTime);

            // 应用音量
            float vol = source->getVolume();
            for (int i = 0; i < sourcePCM.size(); ++i) {
                output[i] += sourcePCM[i] * vol;
            }

            // 应用 Pan（左右声道平衡）
            float pan = source->getPan();  // -1.0 (左) ~ 1.0 (右)
            applyPan(output, pan, m_numChannels);

            // 应用 EQ
            applyEQ(output, source->getEQSettings());

            // 应用 Reverb
            if (source->hasReverb()) {
                applyReverb(output, source->getReverbSettings());
            }
        }

        // 总线音量
        for (int i = 0; i < m_numChannels * BUFFER_FRAMES; ++i) {
            output[i] *= m_volume;
        }

        // 防止削波（Clipping）
        softClip(output, m_numChannels * BUFFER_FRAMES);
    }

private:
    std::string m_name;
    int m_numChannels;
    std::vector<float> m_buffer;
    std::vector<AudioSource*> m_sources;
    float m_volume;

    void applyPan(float* output, float pan, int channels) {
        // 等功率 Pan 算法
        float leftGain = cos(pan * 0.25f * PI);
        float rightGain = sin(pan * 0.25f * PI);
        for (int i = 0; i < BUFFER_FRAMES; ++i) {
            if (channels == 2) {
                output[i * 2] *= leftGain;
                output[i * 2 + 1] *= rightGain;
            }
        }
    }

    void softClip(float* data, int samples) {
        const float threshold = 0.95f;
        for (int i = 0; i < samples; ++i) {
            if (data[i] > threshold) data[i] = threshold;
            if (data[i] < -threshold) data[i] = -threshold;
        }
    }
};

音频流式播放

背景音乐和环境音通常很长（几分钟到几十分钟），不能全部加载到内存。流式播放边解码边播放，只保留一小部分缓冲。

class AudioStream {
public:
    AudioStream(const std::string& filePath)
        : m_filePath(filePath), m_bufferSize(4096) {
        // 打开文件
        m_file.open(filePath, std::ios::binary);
        if (!m_file.is_open()) {
            throw std::runtime_error("Cannot open audio file: " + filePath);
        }

        // 初始化解码器（以 Ogg Vorbis 为例）
        m_decoder = createDecoder(filePath);

        // 创建双缓冲区
        m_buffers[0].resize(m_bufferSize);
        m_buffers[1].resize(m_bufferSize);
        m_currentBuffer = 0;

        // 预填充第一个缓冲区
        fillBuffer(m_currentBuffer);
    }

    // 获取 PCM 数据（每帧调用）
    std::vector<float> getPCM(float deltaTime) {
        std::vector<float> output;
        int framesNeeded = static_cast<int>(SAMPLE_RATE * deltaTime);

        while (output.size() < framesNeeded) {
            // 检查当前缓冲区是否耗尽
            if (m_bufferOffset >= m_bufferSize) {
                // 切换到下一个缓冲区
                m_currentBuffer = 1 - m_currentBuffer;
                m_bufferOffset = 0;

                // 异步填充下一个缓冲区
                fillBufferAsync(m_currentBuffer);
            }

            // 从当前缓冲区复制数据
            int copyFrames = std::min(framesNeeded - output.size(),
                                      m_bufferSize - m_bufferOffset);
            output.insert(output.end(),
                         m_buffers[m_currentBuffer].begin() + m_bufferOffset,
                         m_buffers[m_currentBuffer].begin() + m_bufferOffset + copyFrames);
            m_bufferOffset += copyFrames;
        }

        return output;
    }

private:
    std::string m_filePath;
    std::ifstream m_file;
    Decoder* m_decoder;
    std::vector<std::vector<float>> m_buffers;
    int m_currentBuffer = 0;
    int m_bufferOffset = 0;
    int m_bufferSize;

    void fillBuffer(int bufferIndex) {
        // 同步填充（阻塞）
        m_decoder->decode(m_buffers[bufferIndex].data(), m_bufferSize);
    }

    void fillBufferAsync(int bufferIndex) {
        // 异步填充（后台线程）
        m_decodeQueue.push([this, bufferIndex]() {
            m_decoder->decode(m_buffers[bufferIndex].data(), m_bufferSize);
        });
    }
};

程序化音频生成

有些声音不适合用录音文件——爆炸、碰撞、武器射击的声音可以通过数学公式实时合成。程序化音频的优势是：零磁盘空间、无限变体、与物理状态联动。

// 程序化音频：用数学公式生成各种声音
class ProceduralAudio {
public:
    // 白噪声（用于爆炸、风声）
    static std::vector<float> generateNoise(int numSamples) {
        std::vector<float> samples(numSamples);
        for (int i = 0; i < numSamples; ++i) {
            samples[i] = (rand() / (float)RAND_MAX) * 2.0f - 1.0f;
        }
        return samples;
    }

    // 正弦波（用于测试音、电子音效）
    static std::vector<float> generateSine(float frequency, int numSamples,
                                           float sampleRate = 48000.0f) {
        std::vector<float> samples(numSamples);
        for (int i = 0; i < numSamples; ++i) {
            samples[i] = sinf(2.0f * PI * frequency * i / sampleRate);
        }
        return samples;
    }

    // 爆炸声：噪声 + 指数衰减包络
    static std::vector<float> generateExplosion(int numSamples) {
        auto noise = generateNoise(numSamples);
        std::vector<float> output(numSamples);

        for (int i = 0; i < numSamples; ++i) {
            float envelope = expf(-i / (float)(numSamples * 0.1f));  // 10% 衰减
            output[i] = noise[i] * envelope;
        }
        return output;
    }

    // 激光声：频率递减的正弦波（Chirp）
    static std::vector<float> generateLaser(int numSamples) {
        std::vector<float> output(numSamples);
        float startFreq = 2000.0f;
        float endFreq = 200.0f;

        for (int i = 0; i < numSamples; ++i) {
            float t = (float)i / numSamples;
            float freq = startFreq + (endFreq - startFreq) * t;
            float envelope = expf(-t * 3.0f);  // 包络衰减
            output[i] = sinf(2.0f * PI * freq * t * numSamples / 48000.0f) * envelope;
        }
        return output;
    }

    // 碰撞声：带调谐的噪声脉冲
    static std::vector<float> generateImpact(float hitFrequency, int numSamples) {
        std::vector<float> output(numSamples);
        for (int i = 0; i < numSamples; ++i) {
            float t = (float)i / numSamples;
            float envelope = expf(-t * 8.0f);  // 快速衰减
            float tone = sinf(2.0f * PI * hitFrequency * t * numSamples / 48000.0f);
            float noise = ((rand() / (float)RAND_MAX) * 2.0f - 1.0f) * 0.3f;
            output[i] = (tone * 0.7f + noise * 0.3f) * envelope;
        }
        return output;
    }
};

程序化音频 vs 预录制音频

特性 程序化音频 预录制音频
内存占用几乎为零高（每段音效 1-10MB）
音质电子感强，不够真实真实、细腻
变体无限（随机参数）有限（需要录制多个变体）
与物理联动天然支持（频率/音量随碰撞力度变化）需要多采样切换
CPU 消耗每帧 ~0.01ms解码 ~0.1-0.5ms
典型用途UI 音、科幻武器、环境氛围对话、音乐、写实枪声

完整示例：音频管理器

class AudioManager {
public:
    void init() {
        FMOD::System_Create(&m_system);
        m_system->init(128, FMOD_INIT_NORMAL, nullptr);  // 128 个 Voice
        m_system->createChannelGroup("Master", &m_master);
        m_system->createChannelGroup("Music", &m_musicBus);
        m_system->createChannelGroup("SFX", &m_sfxBus);
        m_system->createChannelGroup("Ambient", &m_ambientBus);
    }

    // 播放音效（自动管理 Voice）
    void playSound(const std::string& clip, const Vec3* position = nullptr) {
        FMOD::Sound* sound = m_clips[clip];
        if (!sound) return;

        FMOD::Channel* channel = nullptr;
        m_system->playSound(sound, m_sfxBus, false, &channel);

        if (position) {
            // 3D 空间音频
            FMOD_VECTOR pos = {position->x, position->y, position->z};
            channel->set3DAttributes(&pos, nullptr);
            channel->setVolume(0.8f);
        } else {
            // 2D 音效（UI、背景音乐等）
            channel->setVolume(1.0f);
        }
    }

    // 播放背景音乐（循环）
    void playMusic(const std::string& clip) {
        FMOD::Sound* sound = m_clips[clip];
        if (!sound) return;
        sound->setMode(FMOD_LOOP_NORMAL);
        m_system->playSound(sound, m_musicBus, false, nullptr);
    }

    // 暂停/恢复所有音频
    void pauseAll(bool paused) {
        FMOD::ChannelGroup* groups[] = {m_master, m_musicBus, m_sfxBus, m_ambientBus};
        for (auto* g : groups) {
            FMOD::Channel* ch = nullptr;
            g->getChannel(0, &ch);
            if (ch) ch->setPaused(paused);
        }
    }

    void update() {
        m_system->update();
    }

    ~AudioManager() {
        for (auto& [name, sound] : m_clips) {
            sound->release();
        }
        m_master->release();
        m_musicBus->release();
        m_sfxBus->release();
        m_ambientBus->release();
        m_system->close();
        m_system->release();
    }

private:
    FMOD::System* m_system = nullptr;
    FMOD::ChannelGroup* m_master = nullptr;
    FMOD::ChannelGroup* m_musicBus = nullptr;
    FMOD::ChannelGroup* m_sfxBus = nullptr;
    FMOD::ChannelGroup* m_ambientBus = nullptr;
    std::unordered_map<std::string, FMOD::Sound*> m_clips;
};

本章小结

本章核心知识点回顾：
  1. 音频架构：源文件 → 解码 → Voice 管理 → 3D 空间化 → 混音总线 → 主输出。每层都可独立优化。
  2. FMOD/Wwise 集成：初始化系统、加载音频、播放 2D/3D 声音、管理 Channel Group。
  3. 3D 空间音频：距离衰减（Reverse/Linear/Exponential）、多普勒效应、HRTF 头相关传输函数。
  4. 混音：等功率 Pan、EQ 滤波、Reverb 卷积。主总线需要 Soft Clip 防止削波。
  5. 流式播放：双缓冲异步解码，内存占用从全量 PCM 降至 ~0.35MB 缓冲。
  6. 程序化音频：噪声、正弦波、Chirp 等数学函数生成音效。零内存、无限变体、与物理联动。
