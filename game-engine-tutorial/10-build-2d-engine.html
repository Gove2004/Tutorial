Sprite 渲染：纹理图集、批处理与实例化

2D 引擎的核心是高效的 Sprite 渲染。渲染成千上万个 Sprite 时，最常见的性能瓶颈是 Draw Call 的数量。解决方案是批处理——将多个 Sprite 合并在一次 GPU 调用中渲染。

纹理图集（Texture Atlas）

// 纹理图集：将多张小图合并到大图中，减少切换纹理的次数
class TextureAtlas {
public:
    TextureAtlas(int width, int height)
        : m_width(width), m_height(height) {
        // 创建空白画布
        glGenTextures(1, &m_textureID);
        glBindTexture(GL_TEXTURE_2D, m_textureID);
        glTexImage2D(GL_TEXTURE_2D, 0, GL_RGBA, width, height, 0,
                     GL_RGBA, GL_UNSIGNED_BYTE, nullptr);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_NEAREST);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_NEAREST);
    }

    // 在图集中分配一块区域
    AtlasRegion allocate(const std::string& name, int w, int h) {
        // 尝试插入（简单的行打包算法）
        int x = tryInsert(w, h);
        if (x == -1) {
            throw std::runtime_error("Atlas full: " + std::to_string(m_width) + "x" + std::to_string(m_height));
        }

        AtlasRegion region;
        region.name = name;
        region.x = x;
        region.y = m_currentY;
        region.width = w;
        region.height = h;
        region.uvMin = Vec2(x / (float)m_width, m_currentY / (float)m_height);
        region.uvMax = Vec2((x + w) / (float)m_width, (m_currentY + h) / (float)m_height);

        m_regions[name] = region;
        m_currentY += h + 1;  // 1px 间隙

        return region;
    }

    // 获取 UV 坐标
    Vec2 getUV(const std::string& name, bool isMax) {
        auto it = m_regions.find(name);
        if (it == m_regions.end()) return Vec2(0);
        return isMax ? it->second.uvMax : it->second.uvMin;
    }

private:
    GLuint m_textureID;
    int m_width, m_height;
    int m_currentY = 0;
    std::unordered_map<std::string, AtlasRegion> m_regions;

    int tryInsert(int w, int h) {
        if (m_currentY + h > m_height) return -1;
        // 检查是否与已有区域重叠
        for (const auto& [name, region] : m_regions) {
            if (region.y <= m_currentY && (region.y + region.height) >= m_currentY) {
                if (region.x < w) return -1;  // 重叠
            }
        }
        return m_lastX;
    }
};

Sprite 批渲染器

// 批渲染器：将同纹理的 Sprite 合并为一次 Draw Call
class SpriteBatch {
public:
    SpriteBatch(TextureAtlas& atlas, int maxSprites = 10000)
        : m_atlas(atlas), m_maxSprites(maxSprites) {
        // 创建顶点缓冲区
        glGenBuffers(1, &m_vbo);
        glGenBuffers(1, &m_ibo);
        m_vertexCount = 0;
        m_indexCount = 0;

        // 初始化顶点数据
        m_vertices.reserve(maxSprites * 4);
        m_indices.reserve(maxSprites * 6);
    }

    // 开始一批渲染
    void begin() {
        m_vertexCount = 0;
        m_indexCount = 0;
        m_currentTexture = nullptr;
    }

    // 添加一个 Sprite
    void draw(const Sprite& sprite) {
        // 纹理切换：需要刷新当前批次
        if (sprite.texture != m_currentTexture && m_vertexCount > 0) {
            flush();
            m_currentTexture = sprite.texture;
        }

        // 如果缓冲区已满，强制刷新
        if (m_vertexCount >= m_maxSprites * 4) {
            flush();
        }

        // 添加四边形（2 个三角形，6 个索引）
        float x = sprite.position.x;
        float y = sprite.position.y;
        float w = sprite.size.x;
        float h = sprite.size.y;

        // UV 坐标
        Vec2 uvMin = sprite.atlasRegion.uvMin;
        Vec2 uvMax = sprite.atlasRegion.uvMax;

        // 4 个顶点（位置 + UV + 颜色）
        m_vertices.push_back(Vertex{x, y, uvMin.x, uvMin.y, sprite.color});
        m_vertices.push_back(Vertex{x + w, y, uvMax.x, uvMin.y, sprite.color});
        m_vertices.push_back(Vertex{x + w, y + h, uvMax.x, uvMax.y, sprite.color});
        m_vertices.push_back(Vertex{x, y + h, uvMin.x, uvMax.y, sprite.color});

        // 6 个索引（两个三角形）
        uint32_t baseIndex = m_vertexCount;
        m_indices.push_back(baseIndex);
        m_indices.push_back(baseIndex + 1);
        m_indices.push_back(baseIndex + 2);
        m_indices.push_back(baseIndex);
        m_indices.push_back(baseIndex + 2);
        m_indices.push_back(baseIndex + 3);

        m_vertexCount += 4;
        m_indexCount += 6;
    }

    // 提交到 GPU
    void flush() {
        if (m_vertexCount == 0) return;

        // 上传顶点数据
        glBindBuffer(GL_ARRAY_BUFFER, m_vbo);
        glBufferData(GL_ARRAY_BUFFER, m_vertexCount * sizeof(Vertex),
                     m_vertices.data(), GL_DYNAMIC_DRAW);

        // 上传索引数据
        glBindBuffer(GL_ELEMENT_ARRAY_BUFFER, m_ibo);
        glBufferData(GL_ELEMENT_ARRAY_BUFFER, m_indexCount * sizeof(uint32_t),
                     m_indices.data(), GL_DYNAMIC_DRAW);

        // 绑定纹理
        if (m_currentTexture) {
            glBindTexture(GL_TEXTURE_2D, m_currentTexture->getID());
        }

        // 一次 Draw Call 渲染所有 Sprite
        glDrawElements(GL_TRIANGLES, m_indexCount, GL_UNSIGNED_INT, 0);

        // 重置
        m_vertices.clear();
        m_indices.clear();
        m_vertexCount = 0;
        m_indexCount = 0;
    }

    ~SpriteBatch() {
        glDeleteBuffers(1, &m_vbo);
        glDeleteBuffers(1, &m_ibo);
    }

private:
    TextureAtlas& m_atlas;
    int m_maxSprites;
    GLuint m_vbo, m_ibo;
    std::vector<Vertex> m_vertices;
    std::vector<uint32_t> m_indices;
    Texture* m_currentTexture = nullptr;
};

实例化渲染（Instanced Rendering）

当大量 Sprite 共享同一纹理时，可以使用 GPU 实例化进一步减少 CPU-GPU 数据传输。

// Instanced rendering：每个实例只需要传递位置和 UV 偏移
// 顶点着色器接收实例属性
class InstancedSpriteRenderer {
public:
    void begin() {
        m_instanceCount = 0;
    }

    void drawInstanced(const Sprite& sprite, int count) {
        if (m_instanceCount + count > MAX_INSTANCES) {
            flush();
        }

        for (int i = 0; i < count; ++i) {
            m_instances[m_instanceCount + i].position = sprite.position;
            m_instances[m_instanceCount + i].uvMin = sprite.uvMin;
            m_instances[m_instanceCount + i].uvMax = sprite.uvMax;
            m_instances[m_instanceCount + i].color = sprite.color;
            m_instances[m_instanceCount + i].scale = sprite.size;
        }
        m_instanceCount += count;
    }

    void flush() {
        if (m_instanceCount == 0) return;

        glBindBuffer(GL_ARRAY_BUFFER, m_instanceVBO);
        glBufferData(GL_ARRAY_BUFFER, m_instanceCount * sizeof(SpriteInstance),
                     m_instances.data(), GL_DYNAMIC_DRAW);

        // 使用 glDrawElementsInstanced 一次渲染所有实例
        glBindTexture(GL_TEXTURE_2D, m_currentTexture->getID());
        glDrawElementsInstanced(GL_TRIANGLES, INDEX_COUNT, GL_UNSIGNED_INT,
                               nullptr, m_instanceCount);

        m_instanceCount = 0;
    }

private:
    static constexpr int MAX_INSTANCES = 100000;
    SpriteInstance m_instances[MAX_INSTANCES];
    int m_instanceCount = 0;
};

瓦片地图系统：TMX 解析与视锥剔除

2D 游戏（尤其是 Roguelike、平台跳跃游戏）广泛使用瓦片地图（Tilemap）。TMX 是 Tiled Map Editor 使用的 XML 格式，是业界标准。

TMX 解析器

class TMXParser {
public:
    TileMap parse(const std::string& filePath) {
        TiXmlDocument doc;
        doc.LoadFile(filePath);

        TileMap map;
        TiXmlElement* root = doc.FirstChildElement("map");
        map.tileWidth = root->IntAttribute("tilewidth");
        map.tileHeight = root->IntAttribute("tileheight");
        map.width = root->IntAttribute("width");
        map.height = root->IntAttribute("height");

        // 解析图层
        for (TiXmlElement* layerEl = root->FirstChildElement("layer");
             layerEl; layerEl = layerEl->NextSiblingElement("layer")) {

            TileLayer layer;
            layer.name = layerEl->Attribute("name");
            layer.width = layerEl->IntAttribute("width");
            layer.height = layerEl->IntAttribute("height");

            // 解析 tile 数据（base64 编码）
            TiXmlElement* dataEl = layerEl->FirstChildElement("data");
            TiXmlDocument dataDoc;
            dataDoc.Parse(dataEl->GetText());

            int tileIndex = 0;
            for (TiXmlElement* tileEl = dataDoc.FirstChildElement("tile");
                 tileEl && tileIndex < layer.width * layer.height;
                 tileEl = tileEl->NextSiblingElement("tile"), ++tileIndex) {

                int gid = tileEl->IntAttribute("gid");
                if (gid != 0) {
                    layer.tiles[tileIndex] = gid;
                }
            }
            map.layers.push_back(layer);
        }

        return map;
    }
};

Chunk 加载与相机裁剪

class TilemapRenderer {
public:
    // Chunk 大小：16x16 tiles
    static constexpr int CHUNK_SIZE = 16;

    void render(const TileMap& map, const Camera& camera) {
        // 计算可见的 chunk 范围
        Rect worldView = camera.getWorldView();

        int startChunkX = worldView.min.x / CHUNK_SIZE;
        int endChunkX = (worldView.max.x + CHUNK_SIZE - 1) / CHUNK_SIZE;
        int startChunkY = worldView.min.y / CHUNK_SIZE;
        int endChunkY = (worldView.max.y + CHUNK_SIZE - 1) / CHUNK_SIZE;

        // 只渲染可见 chunk
        for (int cy = startChunkY; cy <= endChunkY; ++cy) {
            for (int cx = startChunkX; cx <= endChunkX; ++cx) {
                ChunkKey key(cx, cy);

                // 懒加载：按需加载 chunk
                if (m_chunks.find(key) == m_chunks.end()) {
                    loadChunk(map, key);
                }

                // 渲染 chunk
                renderChunk(m_chunks[key], cx * CHUNK_SIZE, cy * CHUNK_SIZE);
            }
        }
    }

private:
    void loadChunk(const TileMap& map, const ChunkKey& key) {
        Chunk chunk;
        chunk.tiles.resize(CHUNK_SIZE * CHUNK_SIZE);

        int offsetX = key.x * CHUNK_SIZE;
        int offsetY = key.y * CHUNK_SIZE;

        for (int y = 0; y < CHUNK_SIZE; ++y) {
            for (int x = 0; x < CHUNK_SIZE; ++x) {
                int mapX = offsetX + x;
                int mapY = offsetY + y;
                if (mapX < map.width && mapY < map.height) {
                    chunk.tiles[y * CHUNK_SIZE + x] =
                        map.getTile(mapX, mapY);
                }
            }
        }
        m_chunks[key] = chunk;
    }

    void renderChunk(const Chunk& chunk, int offsetX, int offsetY) {
        m_batch.begin();
        for (int y = 0; y < CHUNK_SIZE; ++y) {
            for (int x = 0; x < CHUNK_SIZE; ++x) {
                int tileID = chunk.tiles[y * CHUNK_SIZE + x];
                if (tileID == 0) continue;  // 空瓦片

                Sprite sprite;
                sprite.position = Vec2(offsetX + x, offsetY + y);
                sprite.size = Vec2(CHUNK_SIZE, CHUNK_SIZE);
                sprite.texture = m_tileset->getTexture(tileID);
                m_batch.draw(sprite);
            }
        }
        m_batch.flush();
    }

    std::unordered_map<ChunkKey, Chunk> m_chunks;
    SpriteBatch m_batch;
};

粒子系统：GPU 粒子与对象池

粒子系统用于火焰、烟雾、魔法效果等。高性能粒子系统需要对象池复用粒子和 GPU 加速渲染。

CPU 粒子系统 + 对象池

class Particle {
public:
    Vec2 position;
    Vec2 velocity;
    Vec4 color;      // RGBA
    float life;      // 剩余生命
    float maxLife;   // 最大生命
    float size;
    bool alive;
};

class ParticleSystem {
public:
    ParticleSystem(int maxParticles = 10000)
        : m_maxParticles(maxParticles) {
        // 预分配粒子数组（对象池）
        m_particles.resize(maxParticles);
        for (auto& p : m_particles) {
            p.alive = false;
        }
    }

    // 发射粒子（从池中获取可用粒子）
    void emit(const EmitConfig& config, int count) {
        for (int i = 0; i < count; ++i) {
            Particle* p = getFreeParticle();
            if (!p) break;  // 池已满

            // 初始化粒子
            p->position = config.emitterPosition;
            p->velocity = randomInCircle(config.speed) * config.spread;
            p->color = config.startColor;
            p->life = config.lifetime;
            p->maxLife = config.lifetime;
            p->size = config.startSize;
            p->alive = true;
        }
    }

    // 更新所有粒子
    void update(float dt) {
        for (auto& p : m_particles) {
            if (!p.alive) continue;

            // 物理更新
            p->position += p->velocity * dt;
            p->velocity += config.gravity * dt;
            p->life -= dt;

            // 颜色渐变
            float t = 1.0f - p->life / p->maxLife;
            p->color = lerp(config.startColor, config.endColor, t);
            p->size = lerp(config.startSize, config.endSize, t);

            // 生命结束
            if (p->life <= 0) {
                p->alive = false;
            }
        }
    }

    // 渲染（批处理）
    void render(SpriteBatch& batch) {
        for (auto& p : m_particles) {
            if (!p.alive) continue;
            batch.drawSprite(p.position, Vec2(p.size), p.color);
        }
    }

private:
    Particle* getFreeParticle() {
        for (auto& p : m_particles) {
            if (!p.alive) return &p;
        }
        return nullptr;
    }

    std::vector<Particle> m_particles;
    int m_maxParticles;
};

脚本集成：Lua 绑定

脚本语言让策划和设计师可以在不编译引擎的情况下修改游戏逻辑。Lua 是游戏行业最常用的嵌入脚本语言。

Lua 绑定架构

// 使用 sol2（现代 C++ Lua 绑定库）
#include <sol/sol.hpp>

class ScriptingEngine {
public:
    void init() {
        m_lua.open_libraries(sol::lib::base, sol::lib::table,
                              sol::lib::math, sol::lib::string);

        // 绑定 C++ 类到 Lua
        m_lua.new_usertype<Vec3>("Vec3",
            "x", &Vec3::x,
            "y", &Vec3::y,
            "z", &Vec3::z,
            sol::constructors<Vec3(), Vec3(float), Vec3(float, float), Vec3(float, float, float)>()
        );

        m_lua.new_usertype<GameObject>("GameObject",
            "name", &GameObject::getName,
            "setPosition", &GameObject::setPosition,
            "getPosition", &GameObject::getPosition,
            "addComponent", &GameObject::addComponent,
            "destroy", sol::destructor
        );

        m_lua.new_usertype<Scene>("Scene",
            "createObject", &Scene::createObject,
            "getObject", &Scene::getObject,
            "destroyObject", &Scene::destroyObject,
            "update", &Scene::update
        );

        // 绑定全局函数
        m_lua["printDebug"] = [](const std::string& msg) {
            Logger::debug(msg);
        };
        m_lua["spawnObject"] = [this](const std::string& prefab, const Vec3& pos) {
            auto obj = m_scene->createObject(prefab);
            obj->setPosition(pos);
            return obj;
        };
    }

    // 执行 Lua 脚本
    void executeScript(const std::string& filePath) {
        sol::protected_function_result result = m_lua.load_file(filePath);
        if (!result.valid()) {
            sol::error err = result;
            Logger::error("Lua error: " + std::string(err.what()));
        }
    }

    // 从 Lua 调用 C++ 回调
    void registerLuaCallback(const std::string& name, sol::function func) {
        m_luaCallbacks[name] = func;
    }

    void onPlayerDeath() {
        auto it = m_luaCallbacks.find("onPlayerDeath");
        if (it != m_luaCallbacks.end()) {
            it->second();  // 调用 Lua 中的 onPlayerDeath 函数
        }
    }

private:
    sol::state m_lua;
    Scene* m_scene;
    std::unordered_map<std::string, sol::function> m_luaCallbacks;
};

Lua 脚本示例

-- 玩家控制脚本（在 Lua 中编写游戏逻辑）
local player = nil
local speed = 200.0

function init()
    player = GameObject:new("Player")
    player:setPosition(Vec3(0, 0, 0))
    player:addComponent("Rigidbody")
    player:addComponent("SpriteRenderer")
end

function update(dt)
    -- 读取输入
    local moveX = Input.GetAxis("Horizontal")
    local moveY = Input.GetAxis("Vertical")

    -- 移动玩家
    local newPos = player:getPosition()
    newPos.x = newPos.x + moveX * speed * dt
    newPos.y = newPos.y + moveY * speed * dt
    player:setPosition(newPos)

    -- 跳跃
    if Input.GetKeyDown(KeyCode.Space) then
        local rb = player:getComponent("Rigidbody")
        rb:addForce(Vec3(0, 500, 0))
    end
end

function onCollision(other)
    printDebug("Hit: " .. other.name)
    if other.name == "Enemy" then
        spawnObject("Explosion", player:getPosition())
        player:destroy()
    end
end

构建流水线：资源编译与平台部署

资源编译管线

class BuildPipeline {
public:
    // 编译所有资源
    void buildAssets(const std::string& sourceDir, const std::string& outputDir) {
        // 1. 编译纹理
        compileTextures(sourceDir + "/textures/", outputDir + "/textures/");

        // 2. 编译模型
        compileModels(sourceDir + "/models/", outputDir + "/models/");

        // 3. 编译音频
        compileAudio(sourceDir + "/audio/", outputDir + "/audio/");

        // 4. 编译场景
        compileScenes(sourceDir + "/scenes/", outputDir + "/scenes/");

        // 5. 生成资源索引
        generateAssetIndex(outputDir);
    }

    // 编译纹理到 GPU 格式
    void compileTextures(const std::string& inputDir, const std::string& outputDir) {
        for (auto& file : scanFiles(inputDir, "*.png")) {
            // 加载 PNG
            int width, height, channels;
            uint8_t* pixels = stbi_load(file.path.c_str(), &width, &height,
                                        &channels, 4);

            // 压缩为 BC7（PC）或 ASTC（移动平台）
            std::string format = (platform == PLATFORM_PC) ? "BC7" : "ASTC_6x6";
            CompressedTexture compressed = compress(pixels, width, height, format);

            // 写入引擎格式文件
            std::string outFile = outputDir + "/" + file.stem + ".tex";
            writeBinaryFile(outFile, {
                {"magic", "TEX1"},
                {"width", width},
                {"height", height},
                {"format", format},
                {"data", compressed.data}
            });

            stbi_image_free(pixels);
        }
    }

    // 打包为平台部署包
    void deploy(const std::string& outputDir, Platform targetPlatform) {
        std::string bundleName = "engine_" + targetPlatform.toString() + ".zip";

        // 1. 编译引擎二进制
        compileEngine(targetPlatform);

        // 2. 复制资源
        copyDirectory(outputDir, "assets/");

        // 3. 打包
        zipBundle(bundleName, {
            "engine.exe",           // Windows 可执行文件
            "assets/",              // 资源目录
            "scripts/",             // Lua 脚本
            "configs/",             // 配置文件
            "data/",                // 存档数据
        });
    }
};

完整引擎启动流程

// 引擎主循环：整合所有模块
class Application {
public:
    int run() {
        // 1. 初始化子系统
        if (!initSystems()) return 1;

        // 2. 加载初始场景
        loadScene("scenes/start.tmx");

        // 3. 主循环
        Timer timer;
        while (m_running) {
            float dt = timer.getDeltaSeconds();

            // 限制最大帧时间（防止跳帧）
            if (dt > 0.05f) dt = 0.05f;

            // 输入处理
            m_input->update(dt);

            // 脚本更新
            m_scripting->update(dt);

            // 物理更新
            m_physics->step(dt);

            // 音频更新
            m_audio->update();

            // 渲染
            renderFrame();

            // 编辑器更新
            if (m_editor) m_editor->update(dt);
        }

        shutdown();
        return 0;
    }

private:
    bool initSystems() {
        // 窗口
        m_window = std::make_unique<Window>(1280, 720, "My 2D Engine");

        // 渲染器
        m_renderer = std::make_unique<Renderer>(m_window->getGLContext());

        // 物理
        m_physics = std::make_unique<PhysicsWorld>();

        // 音频
        m_audio = std::make_unique<AudioManager>();

        // 输入
        m_input = std::make_unique<InputManager>();

        // 脚本
        m_scripting = std::make_unique<ScriptingEngine>();

        // 编辑器
#ifdef ENGINE_EDITOR
        m_editor = std::make_unique<Editor>(m_renderer.get());
#endif

        // 加载默认配置
        loadConfig("configs/engine.json");

        return true;
    }

    void renderFrame() {
        // 清空
        glClearColor(0.1f, 0.1f, 0.15f, 1.0f);
        glClear(GL_COLOR_BUFFER_BIT);

        // 渲染瓦片地图
        m_tilemapRenderer->render(m_currentMap, m_camera);

        // 渲染精灵
        m_spriteBatch->begin();
        for (auto* obj : m_scene->getRenderables()) {
            m_spriteBatch->draw(obj->getSprite());
        }
        m_spriteBatch->flush();

        // 渲染粒子
        for (auto* ps : m_particleSystems) {
            ps->render(*m_spriteBatch);
        }

        // 渲染 UI
        if (m_editor) {
            m_editor->renderUI();
        }

        // 交换缓冲区
        m_window->swapBuffers();
    }

    std::unique_ptr<Window> m_window;
    std::unique_ptr<Renderer> m_renderer;
    std::unique_ptr<PhysicsWorld> m_physics;
    std::unique_ptr<AudioManager> m_audio;
    std::unique_ptr<InputManager> m_input;
    std::unique_ptr<ScriptingEngine> m_scripting;
    std::unique_ptr<Editor> m_editor;

    SpriteBatch m_spriteBatch;
    TilemapRenderer m_tilemapRenderer;
    Camera m_camera;
    Scene* m_currentScene = nullptr;
    bool m_running = true;
};

// 入口点
int main(int argc, char* argv[]) {
    Application app;
    return app.run();
}

关键数字：使用 SpriteBatch 批渲染，100,000 个 Sprite 在 1080p 下稳定 60fps，Draw Call 从 100,000 降至约 10-50（取决于纹理切换次数）。使用实例化渲染可进一步降至 1-5 个 Draw Call。

本章小结

本章核心知识点回顾：
  1. 项目架构：模块化设计 + CMake 构建系统。每个子系统独立编译为静态库，通过链接组合。
  2. Sprite 渲染：纹理图集减少切换，SpriteBatch 合并 Draw Call，实例化渲染处理海量同纹理 Sprite。
  3. 瓦片地图：TMX 解析 + Chunk 懒加载 + 视锥剔除。只渲染相机范围内的瓦片。
  4. 粒子系统：对象池复用粒子，CPU 粒子适合简单效果，GPU 粒子（Compute Shader）适合大规模特效。
  5. 脚本集成：sol2 绑定 C++ 类到 Lua。游戏逻辑用 Lua 编写，引擎核心用 C++ 编写。
  6. 构建流水线：资源编译（纹理压缩、音频重采样）+ 平台部署（打包为可执行文件）。
