第五章：资源管理——加载、流送、缓存

学习目标：理解游戏引擎如何处理所有外部数据——纹理、模型、音频、脚本。掌握 Asset Pipeline 的设计、异步加载架构、引用计数、资源流送和热重载等核心技术。资源管理的质量直接决定引擎的开发体验和运行时性能。

Asset Pipeline：从源文件到引擎格式

Asset Pipeline（资源管线）是引擎处理外部文件的完整流程。艺术家在 Blender/Maya/Photoshop 中创建的源文件，需要经过一系列转换步骤才能被引擎使用。

为什么要 Asset Pipeline？

直接使用源文件有几个严重问题：

  1. 性能：PSD 文件解压后是几 MB 的原始像素，GPU 需要 BC7 压缩格式才能高效采样。
  2. 跨平台：PC 用 BC7/DXT，iOS 用 ASTC，Android 用 ETC2。Pipeline 为每个平台生成对应格式。
  3. 一致性：所有资源经过统一处理，确保分辨率、采样率、压缩格式符合引擎要求。
  4. 元数据：模型的法线、碰撞体、碰撞层等信息需要额外存储，源文件格式不支持。

纹理加载

纹理是引擎中使用最频繁的资源之一。加载纹理不仅仅是读文件那么简单，需要处理多种格式、生成 Mipmap、压缩到 GPU 格式。

纹理加载流程

class TextureLoader {
public:
    Texture load(const std::string& path) {
        // 1. 检查缓存（避免重复加载同一文件）
        auto it = m_cache.find(path);
        if (it != m_cache.end()) {
            it->second->refCount++;
            return *it->second;
        }

        // 2. 读取原始像素数据
        int width, height, channels;
        uint8_t* pixels = stbi_load(path.c_str(), &width, &height,
                                     &channels, STBI_rgb_alpha);
        if (!pixels) {
            throw std::runtime_error("Failed to load texture: " + path);
        }

        // 3. 生成 Mipmap
        generateMipmaps(pixels, width, height, channels);

        // 4. 压缩到 GPU 格式（BC7 / ASTC / ETC2）
        CompressedData compressed = compressToGPUPixelFormat(pixels, width, height);

        // 5. 上传到 GPU
        GLuint glTexture;
        glGenTextures(1, &glTexture);
        glBindTexture(GL_TEXTURE_2D, glTexture);

        // 主纹理
        glTexImage2D(GL_TEXTURE_2D, 0, compressed.internalFormat,
                     compressed.width, compressed.height, 0,
                     compressed.format, GL_UNSIGNED_BYTE, compressed.data);

        // Mipmap 链
        glGenerateMipmap(GL_TEXTURE_2D);

        // 采样设置
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_S, GL_REPEAT);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_WRAP_T, GL_REPEAT);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MIN_FILTER, GL_LINEAR_MIPMAP_LINEAR);
        glTexParameteri(GL_TEXTURE_2D, GL_TEXTURE_MAG_FILTER, GL_LINEAR);

        // 6. 存入缓存
        auto tex = std::make_shared<Texture>(glTexture, width, height, compressed);
        m_cache[path] = tex;

        // 7. 释放原始数据
        stbi_image_free(pixels);

        return *tex;
    }

private:
    void generateMipmaps(uint8_t* pixels, int width, int height, int channels) {
        // 计算 Mipmap 链：1024 → 512 → 256 → ... → 1
        int mipCount = std::floor(std::log2(std::max(width, height))) + 1;

        // 为每个 Mipmap 级别生成数据
        for (int mip = 1; mip < mipCount; ++mip) {
            int mipWidth = width >> mip;
            int mipHeight = height >> mip;
            if (mipWidth < 1 || mipHeight < 1) continue;

            uint8_t* mipPixels = new uint8_t[mipWidth * mipHeight * channels];
            // 双线性插值缩小
            resizeBilinear(pixels, width, height, mipPixels, mipWidth, mipHeight, channels);
            // ... 存储到压缩数据中
            delete[] mipPixels;
        }
    }
};

GLTF 模型加载示例

Mesh loadGLTF(const std::string& path) {
    tinygltf::Model model;
    std::string err, warn;
    tinygltf::TinyGLTF loader;

    // 加载 GLB（二进制格式，推荐）
    bool success = loader.LoadASCIIFromFile(&model, &err, &warn, path);
    if (!success) {
        throw std::runtime_error("Failed to load GLTF: " + err);
    }

    Mesh mesh;

    // 遍历所有网格
    for (size_t i = 0; i < model.meshes.size(); ++i) {
        const auto& gltfMesh = model.meshes[i];

        // 每个 Primitive 是一个子网格
        for (const auto& primitive : gltfMesh.primitives) {
            SubMesh subMesh;

            // 提取顶点属性
            for (const auto& attr : primitive.attributes) {
                const auto& accessor = model.accessors[attr.first];
                const auto& bufferView = model.bufferViews[accessor.bufferView];
                const uint8_t* data = model.buffers[bufferView.buffer].data.data()
                                    + bufferView.byteOffset + accessor.byteOffset;

                if (attr.first == "POSITION") {
                    subMesh.positions.assign(
                        reinterpret_cast<const Vec3*>(data),
                        reinterpret_cast<const Vec3*>(data) + accessor.count
                    );
                } else if (attr.first == "NORMAL") {
                    subMesh.normals.assign(
                        reinterpret_cast<const Vec3*>(data),
                        reinterpret_cast<const Vec3*>(data) + accessor.count
                    );
                } else if (attr.first == "TEXCOORD_0") {
                    subMesh.texCoords.assign(
                        reinterpret_cast<const Vec2*>(data),
                        reinterpret_cast<const Vec2*>(data) + accessor.count
                    );
                }
            }

            // 提取索引
            if (primitive.indices.size() > 0) {
                subMesh.indices = primitive.indices;
            }

            mesh.subMeshes.push_back(subMesh);
        }
    }

    return mesh;
}

异步加载：后台线程 + 完成回调

资源加载是 I/O 密集型操作。如果在主线程同步加载，会导致帧率骤降甚至卡顿。解决方案是异步加载——在后台线程加载资源，加载完成后通知主线程。

class AsyncResourceLoader {
public:
    // 异步加载请求
    LoadRequestID loadAsync(const std::string& path, LoadCallback callback) {
        LoadRequestID id = m_nextRequestId++;

        // 创建加载请求
        auto request = std::make_shared<LoadRequest>(id, path, std::move(callback));
        request->state = RequestState::Queued;

        {
            std::lock_guard<std::mutex> lock(m_mutex);
            m_queue.push(request);
            m_requests[id] = request;
        }

        m_condition.notify_one();  // 唤醒工作线程
        return id;
    }

    // 阻塞等待加载完成
    Resource* waitForCompletion(LoadRequestID id) {
        auto request = m_requests[id];
        while (request->state != RequestState::Complete &&
               request->state != RequestState::Failed) {
            std::this_thread::yield();
        }

        if (request->state == RequestState::Failed) {
            throw std::runtime_error("Resource load failed: " + request->path);
        }

        return request->resource.get();
    }

    // 工作线程：持续从队列中取任务
    void workerThread() {
        while (m_running) {
            std::shared_ptr<LoadRequest> request;
            {
                std::unique_lock<std::mutex> lock(m_mutex);
                m_condition.wait(lock, [&]() {
                    return !m_queue.empty() || !m_running;
                });

                if (m_queue.empty()) continue;
                request = m_queue.front();
                m_queue.pop();
                request->state = RequestState::Loading;
            }

            // 在后台线程执行实际加载
            try {
                if (endsWith(request->path, ".tex")) {
                    request->resource = std::make_shared<Texture>(
                        loadTextureFromFile(request->path)
                    );
                } else if (endsWith(request->path, ".glb")) {
                    request->resource = std::make_shared<Mesh>(
                        loadGLTFFromFile(request->path)
                    );
                } else if (endsWith(request->path, ".wav")) {
                    request->resource = std::make_shared<AudioBuffer>(
                        loadAudioFromFile(request->path)
                    );
                }

                request->state = RequestState::Complete;
            } catch (const std::exception& e) {
                request->error = e.what();
                request->state = RequestState::Failed;
            }

            // 通知主线程（通过回调）
            if (request->callback) {
                request->callback(request);
            }
        }
    }

private:
    enum class RequestState { Queued, Loading, Complete, Failed };

    struct LoadRequest {
        LoadRequestID id;
        std::string path;
        LoadCallback callback;
        RequestState state;
        std::shared_ptr<Resource> resource;
        std::string error;
    };

    std::queue<std::shared_ptr<LoadRequest>> m_queue;
    std::unordered_map<LoadRequestID, std::shared_ptr<LoadRequest>> m_requests;
    std::mutex m_mutex;
    std::condition_variable m_condition;
    bool m_running = true;
    LoadRequestID m_nextRequestId = 0;
};

资源引用计数：Shared Pointer 模式

多个对象可能引用同一个资源（多个角色使用同一张纹理）。引用计数确保资源在所有使用者都离开后被正确释放。

class Resource {
public:
    virtual ~Resource() = default;

    // 引用计数：增加
    void addRef() {
        m_refCount.fetch_add(1, std::memory_order_relaxed);
    }

    // 引用计数：减少，为零时自动删除
    void release() {
        if (m_refCount.fetch_sub(1, std::memory_order_release) == 1) {
            // 最后一个引用释放，删除资源
            std::atomic_thread_fence(std::memory_order_acquire);
            delete this;
        }
    }

    int refCount() const { return m_refCount.load(); }

protected:
    Resource() : m_refCount(1) {}
    std::atomic<int> m_refCount;
};

// 智能包装器
template<typename T>
class ResourcePtr {
public:
    ResourcePtr(T* ptr = nullptr) : m_ptr(ptr) {
        if (ptr) ptr->addRef();
    }

    ResourcePtr(const ResourcePtr& other) : m_ptr(other.m_ptr) {
        if (m_ptr) m_ptr->addRef();
    }

    ~ResourcePtr() {
        if (m_ptr) m_ptr->release();
    }

    ResourcePtr& operator=(const ResourcePtr& other) {
        if (this != &other) {
            if (m_ptr) m_ptr->release();
            m_ptr = other.m_ptr;
            if (m_ptr) m_ptr->addRef();
        }
        return *this;
    }

    T* get() const { return m_ptr; }
    T* operator->() const { return m_ptr; }

private:
    T* m_ptr;
};

资源流送（Streaming）

对于开放世界等大场景，一次性加载所有资源是不可能的（内存不够、加载时间太长）。资源流送技术将资源分成小块，根据玩家位置动态加载和卸载。

class ResourceManager {
public:
    // 请求资源（自动处理流送）
    ResourcePtr<Resource> request(const std::string& path) {
        // 1. 检查内存缓存
        auto it = m_memoryCache.find(path);
        if (it != m_memoryCache.end()) {
            it->second->addRef();
            return ResourcePtr<Resource>(it->second.get());
        }

        // 2. 检查磁盘缓存（已加载但未入内存）
        auto diskIt = m_diskCache.find(path);
        if (diskIt != m_diskCache.end()) {
            // 将资源从磁盘缓存提升到内存缓存
            auto res = diskIt->second;
            m_memoryCache[path] = res;
            res->addRef();
            return ResourcePtr<Resource>(res.get());
        }

        // 3. 异步从磁盘加载
        m_asyncLoader.loadAsync(path, [this](LoadRequest* req) {
            if (req->state == RequestState::Complete) {
                // 加载完成，放入缓存
                m_memoryCache[req->path] = req->resource;
                // 通知等待者
                if (req->callback) req->callback(req);
            }
        });

        // 返回空指针，等待异步加载完成
        return nullptr;
    }

    // 流送管理：卸载远离玩家的资源
    void updateStreaming(const Vec3& playerPosition, float streamingRadius) {
        // 标记超出范围的资源为待卸载
        for (auto& [path, res] : m_memoryCache) {
            if (res->isLoaded() && !res->isInUse()) {
                float dist = distance(playerPosition, res->getPosition());
                if (dist > streamingRadius) {
                    res->setUnloadPriority(HIGH);
                }
            }
        }

        // 卸载高优先级且未被引用的资源
        std::vector<std::string> toUnload;
        for (auto& [path, res] : m_memoryCache) {
            if (res->unloadPriority() == HIGH && res->refCount() == 1) {
                toUnload.push_back(path);
            }
        }

        for (const auto& path : toUnload) {
            m_diskCache[path] = m_memoryCache[path];  // 保存到磁盘缓存
            m_memoryCache[path]->release();
            m_memoryCache.erase(path);
        }
    }

private:
    std::unordered_map<std::string, std::shared_ptr<Resource>> m_memoryCache;
    std::unordered_map<std::string, std::shared_ptr<Resource>> m_diskCache;
    AsyncResourceLoader m_asyncLoader;
};

热重载（Hot Reloading）

热重载是开发效率的神器——修改资源文件后无需重启游戏，引擎自动重新加载。这对于纹理编辑、模型调整、脚本修改至关重要。

class HotReloadManager {
public:
    void update(float deltaTime) {
        // 扫描所有已加载资源的修改时间
        for (auto& [path, meta] : m_loadedResources) {
            auto currentTime = getFileModificationTime(path);

            if (currentTime > meta.lastCheckTime) {
                // 文件被修改了！
                if (shouldReload(path, currentTime)) {
                    reloadResource(path);
                    meta.lastModified = currentTime;
                }
            }
            meta.lastCheckTime = currentTime;
        }
    }

private:
    bool reloadResource(const std::string& path) {
        // 1. 卸载旧资源（保留引用计数）
        auto oldRes = m_loadedResources[path].resource;
        if (oldRes) {
            oldRes->release();
        }

        // 2. 异步加载新版本
        m_asyncLoader.loadAsync(path, [this, oldPath = path](LoadRequest* req) {
            if (req->state == RequestState::Complete) {
                // 3. 替换缓存中的资源
                m_loadedResources[oldPath].resource = req->resource;
                req->resource->addRef();

                // 4. 通知相关系统（渲染、物理等）
                notifyResourceChanged(oldPath, req->resource);
            }
        });

        return true;
    }

    struct ResourceMeta {
        std::shared_ptr<Resource> resource;
        double lastModified;
        double lastCheckTime;
    };

    std::unordered_map<std::string, ResourceMeta> m_loadedResources;
    AsyncResourceLoader m_asyncLoader;
};
热重载的安全边界：不是所有资源都可以热重载。纹理、模型、音频等静态资源可以安全重载，但场景图结构、物理世界配置等涉及运行时状态的资源需要谨慎处理——可能需要重建场景或重置状态。

本章小结

本章核心知识点回顾：
  1. Asset Pipeline：源文件 → 解析 → 压缩/优化 → 引擎格式 → GPU/音频缓冲区。每个环节都影响性能和兼容性。
  2. 纹理加载：不只是读文件，还包括 Mipmap 生成、GPU 格式压缩（BC7/ASTC/ETC2）、采样设置。
  3. 模型加载：GLTF 是现代标准，支持 PBR 材质和骨骼动画。OBJ 最简单但功能有限。
  4. 异步加载：后台线程 + 完成回调 + 加载状态机。避免主线程阻塞，提供 Loading 界面。
  5. 引用计数：多线程安全的原子计数器，确保资源在最后一个使用者离开后被释放。
  6. 资源流送：按玩家位置动态加载/卸载资源块，支撑开放世界的大场景。
  7. 热重载：监控文件修改时间，异步替换资源，大幅提升开发效率。
