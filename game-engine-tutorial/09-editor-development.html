第九章：编辑器开发——GUI、撤销重做与插件系统

学习目标：使用 Dear ImGui 构建即时模式 GUI，实现场景编辑器（视口、层级面板、属性面板）、撤销/重做系统（命令模式）、资源序列化（JSON/Binary）和插件系统。编辑器是游戏引擎最重要的生产力工具——没有编辑器的引擎只是一个库。

即时模式 GUI vs 保留模式 GUI

GUI 框架分为两种范式：即时模式（Immediate Mode）和保留模式（Retained Mode）。它们代表了完全不同的设计哲学。

特性 即时模式 (Dear ImGui) 保留模式 (Qt/WX/WPF)
渲染频率每帧重绘（60fps）只在变化时重绘
状态管理无状态，由调用方维护框架内部管理控件状态
布局代码即布局（命令式）声明式（XML/QML/HTML）
响应速度零延迟（已在上帧绘制）需要事件传播延迟
定制复杂度简单（重写 draw 回调）复杂（需要主题系统）
内存占用~5-10MB~20-50MB
适用场景游戏编辑器、调试工具桌面应用程序

Dear ImGui 基础

Dear ImGui 是游戏引擎编辑器的事实标准 GUI 库。它的核心理念是：每帧调用一次 UI 代码，框架自动处理渲染、输入处理和状态管理。

// 每帧调用：ImGui 会自动处理输入和渲染
void renderEditorUI(EditorContext& ctx) {
    // 主菜单栏
    if (ImGui::BeginMainMenuBar()) {
        if (ImGui::BeginMenu("文件")) {
            if (ImGui::MenuItem("新建场景", "Ctrl+N")) {
                ctx.sceneManager.createNewScene();
            }
            if (ImGui::MenuItem("保存场景", "Ctrl+S")) {
                ctx.sceneManager.saveScene();
            }
            if (ImGui::MenuItem("退出")) {
                ctx.shouldQuit = true;
            }
            ImGui::EndMenu();
        }
        if (ImGui::BeginMenu("视图")) {
            ImGui::MenuItem("3D 视口", nullptr, ctx.showViewport);
            ImGui::MenuItem("层级面板", nullptr, ctx.showHierarchy);
            ImGui::MenuItem("属性面板", nullptr, ctx.showProperties);
            ImGui::MenuItem("控制台", nullptr, ctx.showConsole);
            ImGui::EndMenu();
        }
        ImGui::EndMainMenuBar();
    }

    // 3D 视口
    if (ctx.showViewport) {
        ImGui::Begin("3D 视口", &ctx.showViewport);
        ImVec2 panelSize = ImGui::GetContentRegionAvail();
        ImDrawList* drawList = ImGui::GetWindowDrawList();

        // 渲染 3D 场景到视口
        renderSceneToViewport(ctx.camera, ctx.scene,
                             ImVec2ToVec2(panelSize), drawList);

        //  gizmo 交互（平移/旋转/缩放）
        ctx.gizmo.render(ctx.selectedObject, panelSize);
        ImGui::End();
    }

    // 层级面板
    if (ctx.showHierarchy) {
        ImGui::Begin("层级面板", &ctx.showHierarchy);
        renderHierarchyPanel(ctx.scene, ctx.selectedObject);
        ImGui::End();
    }

    // 属性面板
    if (ctx.showProperties) {
        ImGui::Begin("属性面板", &ctx.showProperties);
        if (ctx.selectedObject) {
            renderObjectProperties(ctx.selectedObject);
        } else {
            ImGui::TextColored(ImVec4(0.5f, 0.5f, 0.5f, 1.0f),
                              "请选择一个对象");
        }
        ImGui::End();
    }
}

场景编辑器：视口、层级与属性

场景编辑器是游戏编辑器的核心，由三个主要面板组成：3D 视口（渲染场景）、层级面板（对象树）和属性面板（选中对象的属性编辑）。

3D 视口

// 3D 视口渲染：将引擎的渲染管线绑定到 ImGui 的窗口
void renderViewport(EditorContext& ctx) {
    ImVec2 size = ImGui::GetContentRegionAvail();

    // 获取视口的 OpenGL/Vulkan/DX 句柄
    unsigned int textureID = ctx.renderer.getViewportTextureID();

    // 绘制全屏纹理（场景渲染结果）
    ImGui::Image((void*)(intptr_t)textureID, size);

    // 获取鼠标在视口中的位置
    ImVec2 mousePos = ImGui::GetMousePos();
    ImVec2 windowPos = ImGui::GetWindowPos();
    Vec2 viewportMouse = Vec2(
        mousePos.x - windowPos.x,
        mousePos.y - windowPos.y
    );

    // 射线拾取：将鼠标 2D 坐标转为 3D 射线
    Ray pickRay = ctx.camera.screenToWorldRay(viewportMouse, size);

    // Gizmo 交互：平移/旋转/缩放选中的对象
    if (ctx.selectedObject) {
        TransformGizmo gizmo;
        gizmo.setCamera(ctx.camera);
        gizmo.setPosition(ctx.selectedObject->getTransform().position);
        gizmo.setRotation(ctx.selectedObject->getTransform().rotation);
        gizmo.setSize(ctx.selectedObject->getBounds().radius * 2.0f);

        if (gizmo.render(pickRay, size)) {
            // 用户拖动了 gizmo，记录撤销状态
            ctx.undoManager.recordUndo(
                ctx.selectedObject,
                "Move " + ctx.selectedObject->getName()
            );
        }
    }

    // 拖放：从资源浏览器拖入视口放置对象
    if (ImGui::IsItemHovered() && ImGui::IsMouseDragging(0)) {
        // 检查是否有拖放数据
        const ImGuiPayload* payload = ImGui::GetDragDropPayload();
        if (payload && payload->IsDataType(ASSET_TYPE)) {
            AssetRef asset = *(AssetRef*)payload->Data;
            Vec3 worldPos = rayPlaneIntersection(pickRay, Plane(Vec3(0,1,0), 0));
            spawnObjectFromAsset(asset, worldPos);
            ImGui::ResetDragDropPayload();
        }
    }
}

void renderHierarchyPanel(Scene* scene, Object*& selected) {
    if (ImGui::BeginTable("Hierarchy", 3, ImGuiTableFlags_Borders)) {
        ImGui::TableSetupColumn("名称", ImGuiTableCellFlags_None, 200);
        ImGui::TableSetupColumn("可见", ImGuiTableCellFlags_None, 60);

        }

void renderObjectNode(Object* obj, Object*& selected, int depth) {
    ImGuiTreeNodeFlags flags = ImGuiTreeNodeFlags_OpenOnArrow
                             | ImGuiTreeNodeFlags_OpenOnDoubleClick;
    if (obj == selected) flags |= ImGuiTreeNodeFlags_Selected;

    bool expanded = ImGui::TreeNodeEx(obj, flags, "%s", obj->getName().c_str());

    // 行内操作
    ImGui::TableNextColumn();
    ImGui::Checkbox("", &obj->isVisible());

    ImGui::TableNextColumn();
    if (obj->isLocked()) {
        ImGui::SmallIcon("lock");
    }

    // 点击选择
    if (ImGui::IsItemClicked()) {
        selected = obj;
    }

    // 递归子节点
    if (expanded) {
        for (auto* child : obj->getChildren()) {
            renderObjectNode(child, selected, depth + 1);
        }
        ImGui::TreePop();
    }
}

撤销/重做：命令模式

撤销/重做是编辑器的标配功能。实现方式是命令模式：每个操作封装为一个 Command 对象，包含 execute() 和 undo() 方法。执行的操作压入栈，撤销时弹出并调用 undo()。

// 命令基类
class ICommand {
public:
    virtual ~ICommand() = default;
    virtual void execute() = 0;
    virtual void undo() = 0;
    virtual const std::string& getName() const = 0;
};

// 撤销管理器
class UndoManager {
public:
    // 执行命令
    void executeCommand(std::shared_ptr<ICommand> cmd) {
        cmd->execute();
        m_undoStack.push_back(std::move(cmd));
        m_redoStack.clear();  // 新操作清除重做栈
    }

    // 撤销
    bool undo() {
        if (m_undoStack.empty()) return false;
        auto cmd = m_undoStack.back();
        cmd->undo();
        m_redoStack.push_back(std::move(cmd));
        m_undoStack.pop_back();
        return true;
    }

    // 重做
    bool redo() {
        if (m_redoStack.empty()) return false;
        auto cmd = m_redoStack.back();
        cmd->execute();
        m_undoStack.push_back(std::move(cmd));
        m_redoStack.pop_back();
        return true;
    }

    // 批量命令（合并为一次撤销）
    class BatchCommand : public ICommand {
    public:
        BatchCommand(const std::string& name) : m_name(name) {}

        void add(std::shared_ptr<ICommand> cmd) {
            m_commands.push_back(std::move(cmd));
        }

        void execute() override {
            for (auto& cmd : m_commands) cmd->execute();
        }

        void undo() override {
            for (auto it = m_commands.rbegin(); it != m_commands.rend(); ++it) {
                (*it)->undo();
            }
        }

        const std::string& getName() const override { return m_name; }

    private:
        std::string m_name;
        std::vector<std::shared_ptr<ICommand>> m_commands;
    };

private:
    std::vector<std::shared_ptr<ICommand>> m_undoStack;
    std::vector<std::shared_ptr<ICommand>> m_redoStack;
    static constexpr size_t MAX_STACK_SIZE = 1000;  // 最多 1000 步
};

具体命令示例

// 移动对象命令
class MoveObjectCommand : public ICommand {
public:
    MoveObjectCommand(Object* obj, Vec3 oldPos, Vec3 newPos)
        : m_obj(obj), m_oldPos(oldPos), m_newPos(newPos) {}

    void execute() override {
        m_obj->getTransform().position = m_newPos;
    }

    void undo() override {
        m_obj->getTransform().position = m_oldPos;
    }

    const std::string& getName() const override {
        return "Move " + m_obj->getName();
    }

private:
    Object* m_obj;
    Vec3 m_oldPos;
    Vec3 m_newPos;
};

// 添加对象命令
class AddObjectCommand : public ICommand {
public:
    AddObjectCommand(Scene* scene, std::shared_ptr<Object> obj)
        : m_scene(scene), m_obj(std::move(obj)) {}

    void execute() override {
        m_scene->addObject(m_obj);
    }

    void undo() override {
        m_scene->removeObject(m_obj);
    }

    const std::string& getName() const override {
        return "Add " + m_obj->getName();
    }

private:
    Scene* m_scene;
    std::shared_ptr<Object> m_obj;
};

序列化：场景持久化

序列化将内存中的场景对象转换为可存储的格式（JSON/XML/Binary）。编辑器需要保存和加载场景，运行时也需要加载关卡数据。

JSON 序列化

// 使用 nlohmann/json 库
#include <nlohmann/json.hpp>
using json = nlohmann::json;

void serializeSceneToJson(const Scene& scene, std::string& outputPath) {
    json j;
    j["version"] = "1.0";
    j["name"] = scene.getName();

    // 序列化所有根对象
    json objects = json::array();
    for (auto* obj : scene.getRootObjects()) {
        objects.push_back(serializeObject(*obj));
    }
    j["objects"] = objects;

    // 序列化相机
    j["camera"] = {
        {"position", {scene.getCamera().position.x,
                      scene.getCamera().position.y,
                      scene.getCamera().position.z}},
        {"fov", scene.getCamera().fov}
    };

    // 写入文件
    std::ofstream out(outputPath);
    out << j.dump(4);  // 缩进 4 空格
    out.close();
}

json serializeObject(const Object& obj) {
    json j;
    j["name"] = obj.getName();
    j["type"] = obj.getTypeName();

    // 序列化变换
    const Transform& t = obj.getTransform();
    j["transform"] = {
        {"position", {t.position.x, t.position.y, t.position.z}},
        {"rotation", {t.rotation.x, t.rotation.y, t.rotation.z, t.rotation.w}},
        {"scale", {t.scale.x, t.scale.y, t.scale.z}}
    };

    // 序列化组件
    json components = json::array();
    for (auto* comp : obj.getComponents()) {
        components.push_back(serializeComponent(*comp));
    }
    j["components"] = components;

    // 递归序列化子对象
    json children = json::array();
    for (auto* child : obj.getChildren()) {
        children.push_back(serializeObject(*child));
    }
    if (!children.empty()) j["children"] = children;

    return j;
}

Binary 序列化（性能优先）

// Binary 格式：更小、更快，但不易调试
class BinarySerializer {
public:
    void serializeSceneToFile(const Scene& scene, const std::string& path) {
        std::ofstream file(path, std::ios::binary);

        // 文件头
        file.write("ENGS", 4);       // 引擎签名
        file.write(reinterpret_cast<const char*>(&scene.getVersion()), 4);

        // 对象数量
        uint32_t count = scene.getRootObjectCount();
        file.write(reinterpret_cast<const char*>(&count), 4);

        // 序列化每个对象
        for (uint32_t i = 0; i < count; ++i) {
            serializeObjectToFile(*scene.getRootObject(i), file);
        }

        file.close();
    }

private:
    void serializeObjectToFile(const Object& obj, std::ofstream& file) {
        // 类型 ID
        uint32_t typeId = getObjectTypeId(obj.getTypeName());
        file.write(reinterpret_cast<const char*>(&typeId), 4);

        // 名称（变长）
        std::string name = obj.getName();
        uint32_t nameLen = name.size();
        file.write(reinterpret_cast<const char*>(&nameLen), 4);
        file.write(name.c_str(), nameLen);

        // 变换
        const Transform& t = obj.getTransform();
        file.write(reinterpret_cast<const char*>(&t.position), sizeof(Vec3));
        file.write(reinterpret_cast<const char*>(&t.rotation), sizeof(Quat));
        file.write(reinterpret_cast<const char*>(&t.scale), sizeof(Vec3));

        // 组件
        uint32_t compCount = obj.getComponentCount();
        file.write(reinterpret_cast<const char*>(&compCount), 4);
        for (uint32_t i = 0; i < compCount; ++i) {
            serializeComponentToFile(*obj.getComponent(i), file);
        }

        // 子对象
        uint32_t childCount = obj.getChildCount();
        file.write(reinterpret_cast<const char*>(&childCount), 4);
        for (uint32_t i = 0; i < childCount; ++i) {
            serializeObjectToFile(*obj.getChild(i), file);
        }
    }
};

插件系统：动态加载与接口设计

插件系统让引擎可扩展——第三方开发者可以添加新功能而不修改引擎核心代码。关键设计要素：动态库加载、接口抽象、生命周期管理。

// 插件接口：所有插件必须实现
class IPlugin {
public:
    virtual ~IPlugin() = default;
    virtual const char* getName() const = 0;
    virtual const char* getVersion() const = 0;
    virtual void onLoad() = 0;          // 引擎加载插件时调用
    virtual void onUnload() = 0;        // 引擎卸载插件时调用
    virtual void onUpdate(float dt) = 0; // 每帧调用
    virtual void onRenderUI() = 0;      // 编辑器 UI 调用
};

// 插件管理器
class PluginManager {
public:
    void loadPlugin(const std::string& path) {
        // 动态加载 DLL/SO
        void* handle = dlopen(path.c_str(), RTLD_NOW);
        if (!handle) {
            throw std::runtime_error("Cannot load plugin: " + std::string(dlerror()));
        }

        // 获取插件工厂函数
        typedef IPlugin* (*CreatePluginFunc)();
        CreatePluginFunc create = (CreatePluginFunc)dlsym(handle, "createPlugin");
        if (!create) {
            dlclose(handle);
            throw std::runtime_error("Plugin missing createPlugin symbol: " + path);
        }

        // 创建插件实例
        auto* plugin = create();
        plugin->onLoad();

        // 注册
        m_plugins[path] = {plugin, handle};
    }

    void unloadAll() {
        for (auto& [path, entry] : m_plugins) {
            entry.plugin->onUnload();
            dlclose(entry.handle);
        }
        m_plugins.clear();
    }

    template<typename T>
    T* getPlugin() {
        for (auto& [path, entry] : m_plugins) {
            if (T* t = dynamic_cast<T*>(entry.plugin)) return t;
        }
        return nullptr;
    }

private:
    struct PluginEntry {
        IPlugin* plugin;
        void* handle;
    };
    std::unordered_map<std::string, PluginEntry> m_plugins;
};

                importAsset(asset);
            }
            if (ImGui::MenuItem("删除")) {
                deleteAsset(asset);
            }
            ImGui::EndPopup();
        }

        ImGui::NextColumn();
    }
    ImGui::Columns(1);
    ImGui::End();
}

本章小结

本章核心知识点回顾：
  1. Dear ImGui：即时模式 GUI，每帧重绘，零状态管理。标准游戏编辑器 GUI 方案。
  2. 场景编辑器三面板：3D 视口（射线拾取 + Gizmo）、层级面板（树节点递归）、属性面板（组件编辑）。
  3. 撤销/重做：命令模式 + 双向栈。BatchCommand 合并多个操作为一步撤销。
  4. 序列化：JSON 适合编辑期（可读、可手动修改），Binary 适合运行时（快速、体积小）。
  5. 插件系统：动态库 + 接口抽象 + 工厂函数。dlopen/dlsym (Linux/macOS) 或 LoadLibrary/FunProc (Windows)。
  6. 资源浏览器：缩略图缓存 + 拖放传输 + 目录导航。
